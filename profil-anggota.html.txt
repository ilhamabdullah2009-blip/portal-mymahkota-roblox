<script>
"use strict";

/* ======================================
   KONFIGURASI SISTEM
====================================== */

const STORE_KEY =
    "MYMAHKOTA_PORTAL_V1";

const SESSION_KEY =
    "MYMAHKOTA_MEMBER_SESSION";

const $ = id =>
    document.getElementById(id);

function normalize(value){
    return String(value ?? "")
        .trim()
        .toUpperCase();
}

function escapeHTML(value){
    return String(value ?? "")
        .replaceAll("&","&amp;")
        .replaceAll("<","&lt;")
        .replaceAll(">","&gt;")
        .replaceAll('"',"&quot;")
        .replaceAll("'","&#39;");
}

function list(value){
    return Array.isArray(value) ? value : [];
}

/* ======================================
   BACA SESSION ANGGOTA
====================================== */

function readSession(){

    try{
        return JSON.parse(
            sessionStorage.getItem(
                SESSION_KEY
            ) || "null"
        );
    }catch(error){
        return null;
    }

}

/* ======================================
   BACA REKOD PORTAL
====================================== */

function readDatabase(){

    try{

        const data = JSON.parse(
            localStorage.getItem(
                STORE_KEY
            ) || "{}"
        );

        return data &&
            typeof data === "object" &&
            !Array.isArray(data)
            ? data : {};

    }catch(error){
        return {};
    }

}

/* ======================================
   NOMBOR BADAN
====================================== */

function getMemberId(record){

    return normalize(
        record?.noBadan ||
        record?.idAnggota ||
        ""
    );

}

/* ======================================
   KESELAMATAN SESSION
====================================== */

function redirectLogin(){

    sessionStorage.removeItem(
        SESSION_KEY
    );

    window.location.replace(
        "anggota.html"
    );

}

function checkMember(){

    const session = readSession();
    const db = readDatabase();

    if(!session || !session.noBadan){
        redirectLogin();
        return null;
    }

    const memberId = normalize(
        session.noBadan
    );

    const member = list(
        db.members
    ).find(record =>
        getMemberId(record) === memberId
    );

    if(
        !member ||
        normalize(member.status) !== "AKTIF"
    ){
        redirectLogin();
        return null;
    }

    return {
        member,
        db
    };

}

/* ======================================
   SET NILAI
====================================== */

function setText(id,value){

    $(id).textContent =
        value || "—";

}

/* ======================================
   PAPAR JADUAL
====================================== */

function renderRows(
    elementId,
    records,
    fields
){

    const target = $(elementId);

    if(!records.length){

        target.innerHTML =
            '<tr><td colspan="' +
            fields.length +
            '" class="empty">' +
            'Belum ada rekod.' +
            '</td></tr>';

        return;
    }

    target.innerHTML = records.map(
        record => {

            const cells = fields.map(
                field => {

                    const value =
                        record[field] ?? "—";

                    return (
                        "<td>" +
                        escapeHTML(value) +
                        "</td>"
                    );

                }
            ).join("");

            return "<tr>" +
                cells +
                "</tr>";

        }
    ).join("");

}

/* ======================================
   PAPAR PROFIL ANGGOTA
====================================== */

function loadProfile(){

    const result = checkMember();

    if(!result){
        return;
    }

    const {member,db} = result;

    const memberId =
        getMemberId(member);

    /* PROFIL UTAMA */

    setText(
        "profileId",
        memberId
    );

    setText(
        "profileName",
        member.nama
    );

    setText(
        "profileRank",
        member.pangkat
    );

    setText(
        "profileStatus",
        member.status
    );

    $("profileStatus").classList.toggle(
        "inactive",
        normalize(member.status) !== "AKTIF"
    );

    /* MAKLUMAT ANGGOTA */

    const fields = {
        infoName:member.nama,
        infoId:memberId,
        infoRank:member.pangkat,
        infoPosition:member.jawatan,
        infoDepartment:member.jabatan,
        infoUnit:member.unit,
        infoJoinDate:member.tarikhMenyertai,
        infoStatus:member.status
    };

    Object.entries(fields).forEach(
        ([element,value]) => {
            setText(element,value);
        }
    );

    /* REKOD PENTADBIRAN */

    const recordsFor = module =>
        list(db[module]).filter(
            record =>
                getMemberId(record) ===
                memberId
        );

    /* SIJIL */

    renderRows(
        "certificateRows",
        recordsFor("certificates"),
        [
            "rujukan",
            "nama",
            "tarikh",
            "status"
        ]
    );

    /* DOKUMEN */

    renderRows(
        "documentRows",
        recordsFor("documents"),
        [
            "rujukan",
            "tajuk",
            "jenis",
            "tarikh"
        ]
    );

    /* KEMAHIRAN */

    renderRows(
        "skillRows",
        recordsFor("skills"),
        [
            "rujukan",
            "nama",
            "kategori",
            "tahap"
        ]
    );

    /* PRESTASI */

    renderRows(
        "performanceRows",
        recordsFor("performance"),
        [
            "tempoh",
            "jam",
            "sasaran",
            "markah",
            "kod"
        ]
    );

    /* SEJARAH PERKHIDMATAN */

    renderRows(
        "serviceRows",
        recordsFor("serviceRecords"),
        [
            "tarikh",
            "jenis",
            "tajuk",
            "keterangan"
        ]
    );

}

/* ======================================
   LOG KELUAR
====================================== */

$("logoutLink").addEventListener(
    "click",
    function(event){

        event.preventDefault();

        if(
            !confirm(
                "Adakah anda mahu log keluar?"
            )
        ){
            return;
        }

        sessionStorage.removeItem(
            SESSION_KEY
        );

        window.location.href =
            "anggota.html";

    }
);

/* ======================================
   PENGESANAN PERUBAHAN
====================================== */

window.addEventListener(
    "storage",
    function(event){

        if(event.key === STORE_KEY){
            loadProfile();
        }

    }
);

window.addEventListener(
    "focus",
    loadProfile
);

document.addEventListener(
    "visibilitychange",
    function(){

        if(!document.hidden){
            loadProfile();
        }

    }
);

/* ======================================
   MULAKAN SISTEM
====================================== */

loadProfile();
